import {Container, Stack} from '@sentry/scraps/layout';
import {Text} from '@sentry/scraps/text';

import {ProblemSection} from 'sentry/components/events/problemSection';
import * as Storybook from 'sentry/stories';

Callers supply the summary, ordered fields, and optional supporting content. Data
loading, formatting, permissions, and the collapsible section remain with the caller.
The component does not accept an event, occurrence, or issue type.
