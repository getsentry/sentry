export const documentation = import('!!type-loader!@sentry/scraps/menu');

Use `Menu.Root` to build an action menu with a custom layout. Compose its parts in JSX: put items in `Menu.List`, with an optional `Menu.Header` above it and `Menu.Footer` below it. The header and footer remain visible when the list scrolls.

To filter items, place `Menu.Search` inside `Menu.Header`. Search uses each item's `label`; provide `textValue` when the label contains custom content.

For an action menu defined by an array of items, use `DropdownMenu`. For choosing a value, use `CompactSelect`.

## Group

Use `Menu.Section` and `Menu.SectionLabel` to group related actions. Add `Menu.SectionSeparator` between groups when a divider helps distinguish them.

## Checkbox Items

Use `Menu.CheckboxItem` for options that can be turned on or off. Pass `checked` and update it in `onChange`. Selecting one keeps the menu open.

export function CheckboxItemsDemo() {
  const [includeDetails, setIncludeDetails] = useState(true);
  const [includeHistory, setIncludeHistory] = useState(false);
  return (
    <Menu.Root>
      <Menu.Trigger>View options</Menu.Trigger>
      <Menu.Content>
        <Menu.List aria-label="View options">
          <Menu.CheckboxItem
            label="Include details"
            checked={includeDetails}
            onChange={setIncludeDetails}
          />
          <Menu.CheckboxItem
            label="Include history"
            checked={includeHistory}
            onChange={setIncludeHistory}
          />
        </Menu.List>
      </Menu.Content>
    </Menu.Root>
  );
}
