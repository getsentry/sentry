import {useState} from 'react';

import {Menu} from '@sentry/scraps/menu';
import {Text} from '@sentry/scraps/text';

import * as Storybook from 'sentry/stories';

export const documentation = import('!!type-loader!@sentry/scraps/menu');

Use `Menu` when the content needs custom structure. The pieces are composed directly in JSX. `Menu.List` contains the menu items; `Menu.Header` and `Menu.Footer` stay outside its scrolling area. Place `Menu.Search` inside `Menu.Header`; it filters items by their string `label`, or by `textValue` when the label is custom content.

`DropdownMenu` and `CompactSelect` continue to support their existing array APIs.

## Group

Use `Menu.Section` and `Menu.SectionLabel` to group related actions. Add `Menu.SectionSeparator` between groups when a divider helps distinguish them.

<Storybook.Demo>
  <Menu>
    <Menu.Trigger>Grouped actions</Menu.Trigger>
    <Menu.Content>
      <Menu.Header title="Actions">
        <Menu.Search aria-label="Search actions" placeholder="Search actions" />

      <Menu.List aria-label="Grouped actions">

          <Menu.SectionLabel>Manage</Menu.SectionLabel>

          <Menu.ListItem label="Duplicate" onAction={() => {}} />

        </Menu.Section>
      </Menu.List>

        <Text size="sm" variant="muted">
          Choose an action
        </Text>
      </Menu.Footer>

## Checkbox Items

Use `Menu.CheckboxItem` for options that can be turned on or off. Pass `checked` and update it in `onChange`. Selecting one keeps the menu open.

export function CheckboxItemsDemo() {
  const [includeDetails, setIncludeDetails] = useState(true);
  const [includeHistory, setIncludeHistory] = useState(false);
  return (
    <Menu>
      <Menu.Trigger>View options</Menu.Trigger>
      <Menu.Content>
        <Menu.List aria-label="View options">
          <Menu.CheckboxItem
            label="Include details"
            checked={includeDetails}
            onChange={setIncludeDetails}
          />
          <Menu.CheckboxItem
            label="Include history"
            checked={includeHistory}
            onChange={setIncludeHistory}
          />
        </Menu.List>
      </Menu.Content>
    </Menu>
  );
}

</Storybook.Demo>

## SubMenu

Use `Menu.Submenu` for actions that open another menu. Its header can include search for the submenu's items.

<Storybook.Demo>
  <Menu>
    <Menu.Trigger>Actions</Menu.Trigger>
    <Menu.Content>
      <Menu.List aria-label="Actions">
        <Menu.Submenu label="Choose project">
          <Menu.Content>
            <Menu.Header title="Projects">
              <Menu.Search aria-label="Search projects" placeholder="Search projects" />
            </Menu.Header>
            <Menu.List aria-label="Projects">
              <Menu.Section>
                <Menu.SectionLabel>Favorites</Menu.SectionLabel>
                <Menu.ListItem label="Alpha" onAction={() => {}} />
                <Menu.ListItem label="Beta" onAction={() => {}} />
              </Menu.Section>
              <Menu.SectionSeparator />
              <Menu.Section>
                <Menu.SectionLabel>Other projects</Menu.SectionLabel>
                <Menu.ListItem label="Gamma" onAction={() => {}} />
              </Menu.Section>
            </Menu.List>
            <Menu.Footer>
              <Text size="sm" variant="muted">
                Type to filter projects
              </Text>
            </Menu.Footer>
          </Menu.Content>
        </Menu.Submenu>
      </Menu.List>
    </Menu.Content>
  </Menu>
</Storybook.Demo>
