import {useState} from 'react';

import {Container} from '@sentry/scraps/layout';
import {Menu} from '@sentry/scraps/menu';

export const documentation = import('!!type-loader!@sentry/scraps/menu');

`Menu` supplies the shared appearance of dropdown menus and compact selects. It does not create an overlay, hold search or selection state, or assign interaction roles. `DropdownMenu` owns action-menu behavior; `CompactSelect` owns value-selection behavior. Use those controls for application interactions.

The consumer passes state, handlers, refs, and React Aria props to these parts. `Menu.Item` renders the shared row appearance and accepts item behavior through its props. The existing `@sentry/scraps/menuListItem` import re-exports the same component for compatibility.

export function MenuLayoutDemo() {
  const [search, setSearch] = useState('');
  return (
    <Container width="240px">
      <Menu.Header padding="sm lg">
        <Menu.Title>Projects</Menu.Title>
      </Menu.Header>
      <Container padding="xs">
        <Menu.Search
          aria-label="Search projects"
          placeholder="Search projects"
          value={search}
          onChange={event => setSearch(event.target.value)}
        />
      </Container>
      <Container as="ul" margin="0" padding="xs 0">
        <Menu.Section>
          <Container padding="md lg">
            <Menu.SectionTitle>Favorites</Menu.SectionTitle>
          </Container>
          <Menu.SectionItems>
            <Menu.Item label="Alpha" />
            <Menu.Item label="Beta" />
          </Menu.SectionItems>
        </Menu.Section>
      </Container>
      <Menu.Footer padding="md lg">Choose a project</Menu.Footer>
    </Container>
  );
}
