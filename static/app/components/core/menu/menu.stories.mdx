import {useState} from 'react';

import {CompactSelect} from '@sentry/scraps/compactSelect';
import {DropdownMenu} from '@sentry/scraps/dropdownMenu';
import {Menu} from '@sentry/scraps/menu';
import * as Storybook from 'sentry/stories';

`Menu.Header` and `Menu.Footer` sit outside the scrolling collection. Sections need either a visible `Menu.SectionLabel` or an `aria-label`. Place `Menu.Separator` explicitly between sections. `Menu.Item` and `Menu.ListItem` declare collection entries; the hosts render them with React Aria's menu and listbox hooks and the shared `MenuListItem` appearance.

Search belongs to `CompactSelect`. Its `search` prop filters composed options as well as array based options.

export function SearchableComposedSelect() {
  const [value, setValue] = useState();
  return (
    <CompactSelect
      value={value}
      onChange={option => setValue(option.value)}
      search={{placeholder: 'Filter fruit'}}
    >
      <Menu.Section sectionKey="fruit">
        <Menu.SectionLabel>Fruit</Menu.SectionLabel>
        <Menu.ListItem value="apple" label="Apple" />
        <Menu.ListItem value="pear" label="Pear" />
      </Menu.Section>
      <Menu.Footer>Choose one fruit</Menu.Footer>
    </CompactSelect>
  );
}
