import {useState} from 'react';

import {Container} from '@sentry/scraps/layout';

import * as Storybook from 'sentry/stories';
import {OrganizationContext} from 'sentry/utils/organizationContext';
import {useOrganization} from 'sentry/utils/useOrganization';
import {VisualizeTransformations} from './visualizeTransformations';

The span and log toolbars store each series' ordered transformations in URL query parameters.
The editor appears when the organization has the interpolation-and-smoothing feature.

export function TransformationExample() {
  const organization = useOrganization();
  const [transformations, setTransformations] = useState([
    'fill(zero)',
    'smooth(sma,10)',
    'smooth(ema,10,0.25)',
    'smooth(asap,10)',
  ]);
  return (
    <OrganizationContext
      value={{
        ...organization,
        features: [...organization.features, 'explore-interpolation-and-smoothing'],
      }}
    >
      <Container width="280px">
        <VisualizeTransformations
          transformations={transformations}
          onChange={values => setTransformations([...values])}
        />
      </Container>
    </OrganizationContext>
  );
}

## Props

| Prop              | Type                                  | Description                                              |
| ----------------- | ------------------------------------- | -------------------------------------------------------- |
| `transformations` | `readonly string[]`                   | The series' ordered transformation expressions.          |
| `onChange`        | `(values: readonly string[]) => void` | Replaces the list through the parent query-state setter. |

## Accessibility

Each transformation has a numbered group and labeled type, method, settings, and remove controls.
The selects use CompactSelect's keyboard navigation.
